# 二维前缀和

## 区域矩阵的和

**问题描述**：求矩阵中某个区域的和。
**解法**：使用二维前缀和来快速计算任意子矩阵的和。

### 使用vector

```cpp
class NumMatrix {
public:
    vector<vector<int>> matrix;
    // 也可以直接在构造函数中初始化
  	// vector<vector<int>> matrix(n+1, vector<int>(m+1, 0));
    int n,m;
    NumMatrix(vector<vector<int>>& arr) {
        matrix=arr;
        n=matrix.size();
        m=matrix[0].size();
        for(int i=0;i<n;++i){
            for(int j=0;j<m;++j){
                if(i==0&&j==0);
                else if(i==0) matrix[i][j]+=matrix[i][j-1];
                else if(j==0) matrix[i][j]+=matrix[i-1][j];
                else matrix[i][j]+=(matrix[i-1][j]+matrix[i][j-1]-matrix[i-1][j-1]);
            }
        }
    }
  
    int sumRegion(int row1, int col1, int row2, int col2) {
        int ans=matrix[row2][col2];
        if(row1&&col1) ans+=matrix[row1-1][col1-1];
        if(row1) ans-=matrix[row1-1][col2];
        if(col1) ans-=matrix[row2][col1-1];
        return ans;
    }
};
```

### 使用数组

```cpp
int main(){
    int n,m;
    cin>>n>>m;
    int a[n+1][m+1];
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            cin>>a[i][j];
            // 同时输入和计算
            // 不用进行分类
            a[i][j]+=a[i-1][j]+a[i][j-1]-a[i-1][j-1];
        }
    }
    int row1,col1,row2,col2;
    cin>>row1>>col1>>row2>>col2;
    cout<<a[row2][col2]-a[row1-1][col2]-a[row2][col1-1]+a[row1-1][col1-1]<<endl;
    return 0;
}
```
