# 集合

## 结构定义

并查集数据结构的节点和集合定义。

```cpp
typedef struct Node
{
	int val,fa;
	Node(){}
	Node(int val,int fa):val(val),fa(fa){}
}node;

typedef struct Myset
{
	node* p;
	int size,capacity;
	myset(int size,int capa):size(size),capacity(capa){}
}myset;
```

## 初始化与扩展

```cpp
myset* init(int capa=100)
{
	myset* temp=new myset(0,capa);
	temp->p=new node[capa+1];
	for(int i=0;i<=capa;++i) temp->p[i].fa=0;
	return temp;
}

void extend(myset* &S)
{
	node* te=new node[S->capacity*2+1];
	int i;
	for(i=0;i<S->size;++i)
	{
		te[i]=S->p[i];
	}
	S->capacity=S->capacity*2;
	for(;i<=S->capacity;++i) te[i].fa=0;
	delete[] S->p;
	S->p=te;
}
```

## 插入与查找

```cpp
void insert(myset* &S,int x)
{
	if(S->size+1>S->capacity)
	{
		extend(S);
	}
	S->p[x]=node({x,-1});
	S->size++;
}

int find_(myset* &S,int o)
{
	if(o<0 || o>S->capacity || S->p[o].fa==0) return -10;
	while(S->p[o].fa>=0)
		o=S->p[o].fa;
	return o;
}
```

## 合并

```cpp
void union_(myset* &S,int a,int b)
{
	if(a>S->capacity || b>S->capacity)
	{
		cout<<"Out of range!"<<endl;
		return;
	}
	else if(S->p[a].fa==0 || S->p[b].fa==0)
	{
		cout<<"No value!"<<endl;
	}
	else
	{
		int x=find_(S,a);
		int y=find_(S,b);
		if(x==-10 || y==-10)
		{
			cout<<"Error!"<<endl;
			return;
		}
		if(-S->p[x].fa>=-S->p[y].fa)
		{
			S->p[y].fa=x;
			S->p[x].fa--;
		}
		else
		{
			S->p[x].fa=y;
			S->p[y].fa--;
		}
	}
}
```
