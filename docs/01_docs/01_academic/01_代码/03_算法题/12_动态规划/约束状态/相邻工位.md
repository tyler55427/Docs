# 相邻工位

**问题描述**：有 T-1 个工位可以关闭，每个工位有关闭成本，要求关闭 k 个不相邻的工位使得总成本最小。最多能关闭 (n+1)/2 个不相邻工位。

**解法**：动态规划，dp[i][j] 表示前 i 个工位中恰好关闭 j 个且不相邻的最小代价。选第 i 个工位时，i-1 不能选。

```cpp
int main()
{
    int T, k;
    cin >> T >> k;

    int n = T - 1;  // 偏差值个数（可关闭的工位编号 1..n）
    vector<int> cost(n + 1); // 1-based 存储，cost[1]~cost[n]
    for (int i = 1; i <= n; i++)
    {
        cin >> cost[i];
    }

    // 特判：无法关闭 k 个不相邻工位的最大可能值
    // 最多能关闭 (n + 1) / 2 个不相邻工位
    int maxCanClose = (n + 1) / 2;
    if (k > maxCanClose)
    {
        cout << -1 << endl;
        return 0;
    }

    const int INF = 1e9;
    // dp[i][j]：前 i 个工位中恰好关闭 j 个且不相邻的最小代价
    vector<vector<int>> dp(n + 1, vector<int>(k + 1, INF));

    // 初始化：关闭 0 个工位代价为 0
    for (int i = 0; i <= n; i++)
    {
        dp[i][0] = 0;
    }

    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= min(i, k); j++)
        {
            // 不选第 i 个工位
            dp[i][j] = min(dp[i][j], dp[i - 1][j]);

            // 选第 i 个工位（要求 i-1 不能选）
            if (i >= 2)
            {
                dp[i][j] = min(dp[i][j], dp[i - 2][j - 1] + cost[i]);
            }
            else
            {
                // i == 1 且 j == 1 时，可以选第一个工位
                if (j == 1)
                {
                    dp[i][j] = min(dp[i][j], cost[i]);
                }
            }
        }
    }

    int ans = dp[n][k];
    if (ans >= INF)
    {
        cout << -1 << endl;
    }
    else
    {
        cout << ans << endl;
    }

    return 0;
}
```
