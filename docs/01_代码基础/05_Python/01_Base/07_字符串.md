# 字符串

## 字面量

```python
# 单双引号都可以，单引号包含双引号时不用 \
print("This's a Tree.")
print('This\'s a Tree.')

# 原始字符串：r 前缀，所有字符按字面意思输出，不转义
print(r'T\'t')

# 多行字符串：保留所有空格和换行；单引号也可以
print('''
      hello'' world
    1232

      ''')

'''
这里是多行注释
'''
```

## 切片与成员判断

```python
spam = 'Hello World!'
fizz = spam[0:5]                  # 'Hello'
print(fizz)

print('he' in fizz)               # True
print('he' not in fizz)           # False
```

## 字符串插值

```python
# % 格式化
print('My name is %s. I\'m %d years old.' % ('Alice', 20))

# f-string（format 缩写），可以直接放表达式
name = 'AI'
year = 24
print(f'My name is {name}. I am {year + 10} years old')
```

## 大小写与判断方法

```python
spam = 'Hello World!'
print(spam.upper(), spam.lower())
print('h'.islower())              # True

print('sdf'.isalpha())            # 只字母
print('df2'.isalnum())            # 字母 或 数字
print('23243'.isdecimal())        # 只数字
print('     \n'.isspace())        # 只空白
print('Tdskf 8776 Tkk'.istitle()) # 标题样式
```

## 开头 / 结尾

```python
print('The street is broad.'.startswith('The'))   # True
print('The street is broad.'.endswith('d.'))      # True
```

## join / split / partition

```python
spam = ['My', 'name', 'is', 'AI']
print(' '.join(spam))             # 'My name is AI'

spam = 'My namne is AI.'
print(spam.split('e'))            # 默认空白字符
print(spam.partition('n'))        # 只分隔第一次出现；找不到时第一项是整个字符串
```

## 对齐

```python
spam = 'My name is AI.'
print(spam.rjust(20, 'd'))        # 总宽 20，右侧填 'd'
print(spam.ljust(20, 'd'))
print(spam.center(21, '-'))
```

## 去除空白与指定字符

```python
spam = '    Hello world     '
print(spam.strip())               # 去除两端空白
print(spam.lstrip(), rstrip())

spam = 'spamspamHellospamspam'
print(spam.strip('msap'))         # 顺序不重要
```

## 字符 ↔ 数值

```python
print(ord('A'))                   # 65
print(chr(65))                    # 'A'
```
