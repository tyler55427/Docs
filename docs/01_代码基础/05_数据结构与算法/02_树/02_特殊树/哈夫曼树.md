## 辅助函数

选择两个最小的节点、显示树结构。

```cpp
void show(HuffmanTree HT,int s,int e){
	for(int i=s;i<=e;i++){
    	cout << HT[i].weight << " " << HT[i].parent << " " << HT[i].lch << " " << HT[i].rch << endl;
	}
	cout << endl;
}

// 选择两个最小的
void Select(HuffmanTree HT,int n,int &a,int &b){
	int min=10000;
	int j,temp=0;
    for(j=1;j<n;j++){
        if( HT[j].parent == 0 ){
        	if( min>HT[j].weight ){
        		min = HT[j].weight;
        		temp = j;
			}
        }
    }
    a = temp;
    int temp1 = 0;
    min = 10000;
    for(int i=1;i<n;i++){
        if(HT[i].parent == 0 && i!=temp ){
            if( min > HT[i].weight ) {
            	min = HT[i].weight;
            	temp1 = i;
			}
        }
    }
    b = temp1;
}
```

## 构建哈夫曼树

初始化并构建哈夫曼树。

```cpp
// 初始化
void CreatHuffmanTree(HuffmanTree HT,int n){
    if( n<=1 ) return ;
    int m = 2*n-1;
    for(int i=0;i<=m;i++){
    	HT[i].weight=0;
        HT[i].lch=0;
        HT[i].rch=0;
        HT[i].parent=0;
    }
    for(int i=1;i<=n;i++){
        cin >> HT[i].weight;
    }
    show(HT,0,m);
    for(int i=n+1;i<=m;i++){
        int s1=0,s2=0;
        Select(HT,i,s1,s2);
        HT[s1].parent = i;
        HT[s2].parent = i;
        HT[i].lch = s1;
        HT[i].rch = s2;
        HT[i].weight = HT[s1].weight + HT[s2].weight;
    	show(HT,0,m);
    }

}
```

## 哈夫曼编码

从叶子节点向上回溯生成编码，左分支编0，右分支编1。

```cpp
// 哈夫曼编码
void CreatHuffmanCode(HuffmanTree HT,char** &HC,int n){
    char* cd = new char [n];
    cd[n-1] = '/0';
    for(int i=1;i<=n;i++){
        int start = n-1, c = i;
        int f = HT[i].parent;
        while( f != 0 ){
            start--;
            if( HT[f].lch == c ) cd[start] = '0';
            else cd[start] = '1';
            c = f;
            f = HT[f].parent;
        }
        HC[i] = new char[n-start];
        strcpy(HC[i],&cd[start]);
    }
    delete cd;
}
```
